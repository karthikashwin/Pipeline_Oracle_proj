CREATE TABLE DIM_TIME (
    time_key    NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    full_date   DATE,
    day         NUMBER,
    month       NUMBER,
    year        NUMBER,
    quarter     NUMBER
);
INSERT INTO DIM_TIME (full_date, day, month, year, quarter)
SELECT DISTINCT
    DATEID_CLEAN,
    EXTRACT(DAY    FROM DATEID_CLEAN),
    EXTRACT(MONTH  FROM DATEID_CLEAN),
    EXTRACT(YEAR   FROM DATEID_CLEAN),
    CEIL(EXTRACT(MONTH FROM DATEID_CLEAN)/3)
FROM STG_RAW_DAILY
WHERE DATEID_CLEAN IS NOT NULL;

UPDATE STG_RAW_DAILY
SET DATEID_CLEAN =
    CASE 
        WHEN REGEXP_LIKE(DATEID, '^[0-9]{2}-[A-Za-z]{3}-[0-9]{2}$')
            THEN TO_DATE(DATEID, 'DD-MON-RR')
        WHEN REGEXP_LIKE(DATEID, '^[0-9]{2}/[0-9]{2}/[0-9]{4}$')
            THEN TO_DATE(DATEID, 'DD/MM/YYYY')
        WHEN REGEXP_LIKE(DATEID, '^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
            THEN TO_DATE(DATEID, 'YYYY-MM-DD')
        ELSE NULL
    END;
commit;

INSERT INTO FACT_DAILY_PERF (
    time_key, dc_key, product_key, vendor_key, channel_key,
    units_sold, unit_price, discount, binding_cost, revenue, gross_margin
)
SELECT
    t.time_key,
    dc.dc_key,
    p.product_key,
    v.vendor_key,
    ch.channel_key,

    s."TQty",
    s."UPrice",
    s."Dscnt",
    d.BC,
    d.REV,
    (d.REV - d.BC) AS gross_margin
FROM STG_RAW_SALES s
JOIN STG_RAW_DAILY d 
      ON s.PD = d.PD
LEFT JOIN STG_RAW_META m 
      ON s."EdID" = m."publication/ed"
LEFT JOIN DIM_TIME t
      ON t.full_date = d.DATEID_CLEAN
LEFT JOIN DIM_DC dc
      ON dc.station_code = d.STNCODE
LEFT JOIN DIM_PRODUCT p
      ON p.edition_id = s."EdID"
LEFT JOIN DIM_VENDOR v
      ON v.vendor_score = m."vendor/score"
LEFT JOIN DIM_CHANNEL ch
      ON ch.channel_code = s."Chnl";

SELECT
    dc.address AS region,
    p.title AS edition,
    SUM(f.revenue - f.binding_cost) AS gross_margin,
    RANK() OVER (
        ORDER BY SUM(f.revenue - f.binding_cost) DESC
    ) AS rn
FROM FACT_DAILY_PERF f
JOIN DIM_DC dc
    ON f.dc_key = dc.dc_key
JOIN DIM_PRODUCT p
    ON f.product_key = p.product_key
JOIN DIM_TIME dt
    ON f.time_key = dt.time_key
WHERE dt.YEAR = 2024
  AND dt.QUARTER = 4
GROUP BY dc.address, p.title;



SELECT 
    dt.YEAR,
    dt.MONTH,
    v.vendor_score,
    SUM(f.binding_cost) AS total_binding_cost,
    AVG(SUM(f.binding_cost)) OVER (
        PARTITION BY v.vendor_score
        ORDER BY dt.YEAR, dt.MONTH
        ROWS BETWEEN 11 PRECEDING AND CURRENT ROW
    ) AS rolling_12_month_avg
FROM FACT_DAILY_PERF f
JOIN DIM_VENDOR v 
    ON f.vendor_key = v.vendor_key
JOIN DIM_TIME dt 
    ON f.time_key = dt.time_key
GROUP BY 
    dt.YEAR,
    dt.MONTH,
    v.vendor_score
ORDER BY 
    v.vendor_score,
    dt.YEAR,
    dt.MONTH;


SELECT 
    p.title AS edition,
    c.channel_name,
    SUM(f.units_sold) AS total_units,
    ROUND(
        SUM(f.units_sold) * 100 
        / SUM(SUM(f.units_sold)) OVER (PARTITION BY p.title), 
        2
    ) AS pct_of_total
FROM FACT_DAILY_PERF f
JOIN DIM_PRODUCT p 
    ON f.product_key = p.product_key
JOIN DIM_CHANNEL c
    ON f.channel_key = c.channel_key
GROUP BY 
    p.title, c.channel_name
ORDER BY 
    p.title, pct_of_total DESC;


SELECT 
    p.title AS edition,
    SUM(f.revenue) AS total_revenue,
    AVG(f.revenue - f.binding_cost) AS avg_margin
FROM FACT_DAILY_PERF f
JOIN DIM_PRODUCT p
    ON f.product_key = p.product_key
GROUP BY p.title
ORDER BY total_revenue DESC;


SELECT 
    dc.address AS dc_name,
    dc.address AS region,   -- using address as region substitute
    (SUM(f.revenue) - SUM(f.binding_cost)) / NULLIF(SUM(f.revenue), 0) AS profit_margin_pct
FROM FACT_DAILY_PERF f
JOIN DIM_DC dc 
    ON f.dc_key = dc.dc_key
GROUP BY 
    dc.address
HAVING 
    (SUM(f.revenue) - SUM(f.binding_cost)) / NULLIF(SUM(f.revenue), 0) < 0.15
ORDER BY 
    profit_margin_pct ASC;
